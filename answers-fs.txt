实现大文件的存储

xv6通过盘块号在磁盘上得到对应的盘块。一个盘块占用1024字节。

在当前的xv6系统中inode有13个存储盘块号的域。这些盘块号占用4字节
其中12个是直接盘块号，对于小于12*1024字节的文件的读写会依次使用这些直接盘块
其中有一个间接盘块号，通过间接盘块号可以得到一个盘块，盘块内存储了最多1024/4=256个盘块号，这256个盘块将用于存储数据。
所以当前可以存储的最大文件是1024*(12+256)字节。

现在我们的任务是牺牲一个直接盘块，作为二级间接盘块，可存储256*256个盘块号。实现更大文件的存储。


新增一个inode的盘块域
``` c
struct dinode {
  short type;           // File type
  short major;          // Major device number (T_DEVICE only)
  short minor;          // Minor device number (T_DEVICE only)
  short nlink;          // Number of links to inode in file system
  uint size;            // Size of file (bytes)
  uint addrs[NDIRECT+2];   // Data block addresses
};
```

修改直接盘块数，一级间接盘块数，二级间接盘块数的宏定义
``` c
#define NDIRECT 11
#define NINDIRECT (BSIZE / sizeof(uint))
#define NDINDIRECT (NINDIRECT * NINDIRECT)
```

我们发现读写inode的函数都调用了`uint bmap(struct inode *ip, uint bn)`。寻找当前inode中的第bn个盘块的盘块号。
修改主要是添加二级间接盘块的处理。
``` c
static uint
bmap(struct inode *ip, uint bn)
{
  uint addr, *a, *a1;
  struct buf *bp, *bp1;
  
  if(bn < NDIRECT){
    if((addr = ip->addrs[bn]) == 0){
      addr = balloc(ip->dev);
      if(addr == 0)
        return 0;
      ip->addrs[bn] = addr;
    }
    return addr;
  }
  bn -= NDIRECT;

  if(bn < NINDIRECT){
    // Load indirect block, allocating if necessary.
    if((addr = ip->addrs[NDIRECT]) == 0){
      addr = balloc(ip->dev);
      if(addr == 0)
        return 0;
      ip->addrs[NDIRECT] = addr;
    }
    bp = bread(ip->dev, addr);
    a = (uint*)bp->data;
    if((addr = a[bn]) == 0){
      addr = balloc(ip->dev);
      if(addr){
        a[bn] = addr;
        log_write(bp);
      }
    }
    brelse(bp);
    return addr;
  }
  bn -= NINDIRECT;

  if(bn < NDINDIRECT){
    uint d1 = bn/NINDIRECT, d2 = bn%NINDIRECT;

    if((addr = ip->addrs[NDIRECT+1]) == 0){
      addr = balloc(ip->dev);
      if(addr == 0)
        return 0;
      ip->addrs[NDIRECT+1] = addr;
    }
    bp = bread(ip->dev, addr);
    a = (uint*)bp->data;
    if((addr = a[d1]) == 0){
      addr = balloc(ip->dev);
      if(addr){
        a[d1] = addr;
        log_write(bp);
      }
    }
    bp1 = bread(ip->dev, addr);
    a1 = (uint*)bp1->data;
    if ((addr = a1[d2]) == 0) {
      addr = balloc(ip->dev);
      if (addr) {
        a1[d2] = addr;
        log_write(bp1);
      }
    }
    brelse(bp1);
    brelse(bp);
    return addr;
  }

  panic("bmap: out of range");
}
```

软链接的实现  

原本的思路存在错误，我在inode中将类型设置为软链接类型T_SYMLINK，然后用原本存盘块的一个域去存储链接的inode的inum。
对于读写软链接，如果链接的inode是软链接则继续链接到新的节点，为了防止环出现或链接过深。设置了最大链接深度。
但是实际上并非如此，软链接是可以悬挂的指向一个不存在的路径，当这个路径创建后将会生效。

于是我们只能将链接的路径，存储的到软链接的inode中。
具体做法如下：
对于链接old路径到new路径。
先找到new路径中父目录的inode指针dp，同时获取new路径中的要创建的软链接名称name。
创建一个新inode，其指针为ip，写入old路径字符串，然后在dp目录节点中加入(name, ip->inum)目录项

期间错误处理比较麻烦，比如在创建inode后写入失败，或链接失败都需要销毁这个indoe。
另外indoe必须要上锁才能读取到最新内容。

``` c

uint64
sys_symlink(void)
{
  char name[MAXPATH], new[MAXPATH], old[MAXPATH];
  struct inode *ip, *dp;
  
  if(argstr(0, old, MAXPATH) < 0 || argstr(1, new, MAXPATH) < 0)
  return -1;
  
  /*
    new是一个路径，old是一个路径，new末端是一个symlink，指向old的路径

    找到new的父目录dp，末端的名字name，如果无法找到dp，返回-1
    查看dp中是否已经存在name的文件，如果存在，返回-1
    新建一个inode ip，类型为symlink，设置硬链接数为1, 写入old的路径
    在dp的目录中加入文件name，指向ip->inum，失败 需要删除ip
  */
  
  begin_op();
  
  if((dp = nameiparent(new, name, 0)) == 0) {
    end_op();
    return -1;
  }

  if((ip = dirlookup(dp, name, 0)) != 0){
    iput(ip);
    iput(dp);
    end_op();
    return -1;
  }

  if((ip = ialloc(dp->dev, T_SYMLINK)) == 0){
    iput(dp);
    end_op();
    return -1;
  }

  ilock(ip);
  ip->major = 0;
  ip->minor = 0;
  ip->nlink = 1;
  iupdate(ip);

  // 在这个inode上写入old路径
  if (writei(ip, 0, (uint64)old, 0, strlen(old)) != strlen(old)) {
    iput(dp);
    goto bad; // ip已锁
  }
  
  
  // 链接
  // printf("symlink old=%s new=%s\n", old, new);
  // printf("symlink dp->inum=%d dp->type=%d, ip->inum=%d ip->type=%d\n", dp->inum, dp->type, ip->inum, ip->type);
  ilock(dp);
  if(dp->dev != ip->dev || dirlink(dp, name, ip->inum) < 0){
    iunlockput(dp);
    goto bad;
  }

  iunlockput(ip);
  iunlockput(dp);
  end_op();
  return 0;

bad:
  ip->nlink = 0;
  iupdate(ip);
  iunlockput(ip);
  end_op();
  return -1;  
}
```
对于打开软链接，实际上打开的是链接到的文件，通过路径寻找inode的过程中，如果存储的是相对路径，那么就要以当前软链接文件所在目录为起点，绝对路径就以根目录为起点

``` c

// 传入lock的ip，找到并返回ip的链接目标(也是lock的），目标前的经过的inode全部put 
struct inode* 
ifollow(struct inode *ip) {
  // printf("ifollow ip->inum=%d\n", ip->inum);
  struct inode *next;
  
  char path[MAXPATH];
  int i;
  for (i=0; i<MAX_SLINK_DEEP; i++) {
    // ip 已经锁了
    memset(path, 0, MAXPATH);
    readi(ip, 0, (uint64)path, 0, MAXPATH);
    // printf("ip->inum=%d link=%s\n", ip->inum, path);
    if (path[0] == '\0') {
      iunlockput(ip);
      return 0;
    }
    iunlock(ip); //解锁
    next = namei(path, ip);
    iput(ip); // 放下ip的引用
    if (next == 0) return 0;
    ilock(next); // 只有锁了才能读到内容
    // printf("next->inum=%d, next->type=%d\n", next->inum, next->type);
    if (next->type != T_SYMLINK) break;
    ip = next;
  }
  if (i == MAX_SLINK_DEEP) { // ip解锁释放引用
    iunlockput(ip);
    return 0;
  }
  if (next->nlink == 0) {
    iunlockput(next);
    return 0;
  }
  return next;
}
```
